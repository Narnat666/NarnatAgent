// 计算对话的 token 占用
const stats = new Map();
function addTurn(id, tokens) {
  if (!stats.has(id)) stats.set(id, { in: 0, out: 0 });
  const cur = stats.get(id);
  cur.in += tokens.input;
  return `turn ${id}: ${cur.in} tokens`;
}

SQL

sql
SELECT s.name, COUNT(m.id) AS msg_count
FROM sessions s
LEFT JOIN messages m ON m.session_id = s.id
WHERE s.updated_at > NOW() - INTERVAL 7 DAY  -- 最近一周
GROUP BY s.name
ORDER BY msg_count DESC;
